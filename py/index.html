PyScript 실행 안정화

PyScript 로더가 실패하면 아래에 원인을 표시합니다.

from pyscript import Element from pyodide.http import pyfetch RAW_URL = None # 필요 시 "https://raw.githubusercontent.com/{USER}/{REPO}/HEAD/scripts/demo.py" async def run_from_repo(): out = Element("out") try: out.write("[불러오는 중] scripts/demo.py ...\n") try: resp = await pyfetch("scripts/demo.py") if resp.status != 200: raise RuntimeError(f"HTTP {resp.status}") code = await resp.text() except Exception as e1: if RAW_URL: out.write(f"[안내] 로컬 fetch 실패 → RAW로 재시도: {e1}\n") resp2 = await pyfetch(RAW_URL, method="GET") if resp2.status != 200: raise RuntimeError(f"RAW fetch 실패: HTTP {resp2.status}") code = await resp2.text() else: raise ns = {} exec(code, ns, ns) if "main" in ns and callable(ns["main"]): out.write(str(ns["main"]()) + "\n") else: out.write("[안내] demo.py에 main()이 없습니다.\n") except Exception as e: out.write("[에러] PyScript 초기화/네트워크 문제일 수 있습니다.\n") out.write(f"└ 상세: {e}\n") out.write("※ 점검: 1) Pages URL로 열었는지 2) core.js 200 OK인지 3) scripts/demo.py 경로/대소문자\n") def clear_out(): Element("out").clear()
