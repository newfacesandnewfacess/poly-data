레포의 Python 파일을 브라우저에서 실행 (PyScript)

버튼을 클릭하면 scripts/demo.py를 불러와 main()을 실행합니다.

출력

from pyscript import Element from pyodide.http import pyfetch async def run_from_repo(): out = Element("out") out.write("[불러오는 중] ./scripts/demo.py ...\n") try: resp = await pyfetch("scripts/demo.py") # 같은 레포(동일 출처)라 CORS 문제 없음 code = await resp.text() ns = {} exec(code, ns, ns) if "main" in ns and callable(ns["main"]): result = ns["main"]() out.write(str(result) + "\n") else: out.write("[안내] demo.py에 main() 함수가 없습니다.\n") except Exception as e: out.write(f"[에러] {e}\n") def clear_out(): Element("out").clear()
