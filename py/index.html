PyScript · 버전 고정 + 네트워크 Fallback

CORS/경로 문제 시 RAW로 재시도하고, 원인을 아래에 표시합니다.


    

  


  
from pyscript import display
from pyodide.http import pyfetch
from js import document

# 👇 본인 깃허브에 맞게 교체(예: alexkim/myrepo)
RAW_URL = "https://raw.githubusercontent.com//newfacesandnewfacess/poly-data/scripts/demo.py"

def w(msg): display(str(msg), target="out", append=True)

async def run_from_repo():
    try:
        w("[불러오는 중] ./scripts/demo.py ...\\n")
        try:
            resp = await pyfetch("./scripts/demo.py")
            if resp.status != 200:
                raise RuntimeError(f"HTTP {resp.status}")
            code = await resp.text()
        except Exception as e1:
            w(f"[안내] 로컬 fetch 실패 → RAW 재시도: {e1}\\n")
            resp2 = await pyfetch(RAW_URL, method="GET")
            if resp2.status != 200:
                raise RuntimeError(f"RAW fetch 실패: HTTP {resp2.status}")
            code = await resp2.text()

        ns = {}
        exec(code, ns, ns)
        if "main" in ns and callable(ns["main"]):
            w(ns["main"]()); w("\\n")
        else:
            w("[안내] demo.py에 main()이 없습니다.\\n")

    except Exception as e:
        w("[에러] PyScript/네트워크/경로 문제일 수 있습니다.\\n")
        w(f"└ 상세: {e}\\n")
        w("※ 점검: 1) github.io(Pages)로 열기  2) Network에서 core.js 200 OK  3) scripts/demo.py 경로·대소문자\\n")

def clear_out():
    document.getElementById("out").innerText = ""
