from pyscript import display
from pyodide.http import pyfetch
from js import document
import asyncio

def out(target, msg): display(str(msg), target=target, append=True)

async def pip_install(pkg: str):
    import micropip
    await micropip.install(pkg)
    return f"installed: {pkg}"

def clear_out(target):
    document.getElementById(target).innerText = ""

def run_code():
    src = document.getElementById("code").value
    try:
        ns = {}
        exec(src, ns, ns)
    except Exception as e:
        out("out1", f"[에러] {e}\\n")

async def run_from_repo():
    try:
        out("out2", "[불러오는 중] scripts/demo.py ...\\n")
        resp = await pyfetch("scripts/demo.py")
        if resp.status != 200:
            out("out2", f"[에러] HTTP {resp.status}\\n"); return
        code = await resp.text()
        ns = {}
        exec(code, ns, ns)
        if "main" in ns and callable(ns["main"]):
            out("out2", ns["main"]()); out("out2", "\\n")
        else:
            out("out2", "[안내] demo.py에 main()이 없습니다.\\n")
    except Exception as e:
        out("out2", f"[에러] {e}\\n")
